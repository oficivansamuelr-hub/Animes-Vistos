export const Portada = ({ titulo, id }) => {
  const [src, setSrc] = useState(null);

  useEffect(() => {
    const query = id
      ? "query ($id: Int) { Media(id: $id, type: ANIME) { coverImage { extraLarge large } } }"
      : "query ($search: String) { Media(search: $search, type: ANIME) { coverImage { extraLarge large } } }";
    const variables = id ? { id: Number(id) } : { search: titulo };

    fetch("https://graphql.anilist.co", {
      method: "POST",
      headers: { "Content-Type": "application/json", Accept: "application/json" },
      body: JSON.stringify({ query, variables }),
    })
      .then((r) => r.json())
      .then((d) => {
        const c = d && d.data && d.data.Media && d.data.Media.coverImage;
        if (c) setSrc(c.extraLarge || c.large);
      })
      .catch(() => {});
  }, [titulo, id]);

  if (!src) return null;

  return (
    <div style={{ margin: "0 0 24px 0" }}>
      <img
        src={src}
        alt={titulo}
        style={{ width: "100%", maxWidth: "260px", borderRadius: "12px" }}
      />
    </div>
  );
};
